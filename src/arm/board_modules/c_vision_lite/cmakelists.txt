cmake_minimum_required(VERSION 3.13)
project(labmate_lite_probe LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LITE_CXX_DIR "" CACHE PATH "Extracted Paddle-Lite cxx directory")
set(LITE_LIBRARY_DIR "${LITE_CXX_DIR}/lib" CACHE PATH "Paddle-Lite shared API library directory")
option(BUILD_IMAGE_PROBE "Build the OpenCV image input probe" OFF)
if(NOT EXISTS "${LITE_CXX_DIR}/include/paddle_api.h")
  message(FATAL_ERROR "Set LITE_CXX_DIR to the extracted Paddle-Lite cxx directory")
endif()

if(EXISTS "${LITE_LIBRARY_DIR}/libpaddle_light_api_shared.so")
  set(LITE_API_LIBRARY paddle_light_api_shared)
elseif(EXISTS "${LITE_LIBRARY_DIR}/libpaddle_full_api_shared.so")
  set(LITE_API_LIBRARY paddle_full_api_shared)
else()
  message(FATAL_ERROR "No Paddle-Lite shared API library in ${LITE_LIBRARY_DIR}")
endif()

add_executable(smoke_infer smoke_infer.cc)
target_include_directories(smoke_infer PRIVATE "${LITE_CXX_DIR}/include")
target_link_directories(smoke_infer PRIVATE "${LITE_LIBRARY_DIR}")
target_link_libraries(smoke_infer PRIVATE ${LITE_API_LIBRARY} dl pthread)

if(BUILD_IMAGE_PROBE)
  find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
  add_executable(labmate_lite_image_probe image_probe.cc)
  target_include_directories(labmate_lite_image_probe PRIVATE "${LITE_CXX_DIR}/include")
  target_link_directories(labmate_lite_image_probe PRIVATE "${LITE_LIBRARY_DIR}")
  target_link_libraries(labmate_lite_image_probe PRIVATE ${LITE_API_LIBRARY} ${OpenCV_LIBS} dl pthread)
endif()
